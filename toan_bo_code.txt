async function ords(path, method = "GET", body) {
  const r = await fetch(CONFIG.BASE + path, {
    method,
    headers: body !== undefined ? { "Content-Type": "application/json" } : {},
    body: body !== undefined ? JSON.stringify(body) : undefined,
  });
  let j = {};
  try {
    j = await r.json();
  } catch (_) {}
  if (!r.ok) {
    const raw = [j.cause, j.detail, j.message, j.error]
      .filter(Boolean)
      .join("\n");
    const m = /ORA-(\d+):?\s*([^\n]*)/.exec(raw);
    raise(
      m ? -Number(m[1]) : r.status,
      m ? m[2].trim() : j.message || "Lá»—i káº¿t ná»‘i " + r.status,
    );
  }
  return j;
}

function price(st, seat) {
  let p = st.base_price * ({ VIP: 1.25, COUPLE: 2 }[seat.seat_type] || 1);
  const d = new Date(st.start_time);
  if (d.getDay() === 0 || d.getDay() === 6) p *= 1.15;
  if (d.getHours() < 12) p *= 0.8;
  return Math.round(p / 1000) * 1000;
}
function discount(code, sub, st) {

  if (
    p.days &&
    !p.days.includes(

        new Date(st.start_time).getDay()

    )
  )
    raise(-20033, "MÃ£ khÃ´ng Ã¡p dá»¥ng cho ngÃ y chiáº¿u nÃ y");

}

BEGIN

    p_enabled             => TRUE,
    p_schema              => 'CINEMA_SCHEMA',
    p_url_mapping_type    => 'BASE_PATH',
    p_url_mapping_pattern => 'cinema',
    p_auto_rest_auth      => FALSE
  );

  BEGIN
    ORDS.DELETE_MODULE(p_module_name => 'cinema_api');
  EXCEPTION
    WHEN OTHERS THEN NULL;
  END;

  ORDS.DEFINE_MODULE(
    p_module_name    => 'cinema_api',
    p_base_path      => '/',
    p_items_per_page => 200,
    p_status         => 'PUBLISHED',
    p_comments       => 'CineDB REST API synced with js/api.js'
  );

  -- 1. PHIM
  ORDS.DEFINE_TEMPLATE(p_module_name => 'cinema_api', p_pattern => 'movies');
  ORDS.DEFINE_HANDLER(
    p_module_name => 'cinema_api', p_pattern => 'movies', p_method => 'GET',
    p_source_type => 'json/collection',
    p_source      => q'[
      SELECT m.movie_id, m.title, m.genre, m.duration_min, m.rating,
             m.release_date, m.metadata, m.status,
             NVL((SELECT ROUND(AVG(r.rating), 1) FROM review r WHERE r.movie_id = m.movie_id), 0) AS avg_rating,
             (SELECT COUNT(*) FROM review r WHERE r.movie_id = m.movie_id) AS review_count
        FROM movie m
       ORDER BY m.movie_id
    ]'
  );
  BEGIN
  ORDS.DEFINE_HANDLER(
    p_module_name => 'cinema_api',
    p_pattern     => 'movies',
    p_method      => 'POST',
    p_source_type => 'plsql/block',
    p_source      => q'[
      DECLARE
        v_raw    CLOB;
        v_obj    JSON_OBJECT_T;
        v_elem   JSON_ELEMENT_T;
        v_id     NUMBER := NULL;
        v_title  VARCHAR2(200);
        v_genre  VARCHAR2(100);
        v_dur    NUMBER;
        v_rating VARCHAR2(20);
        v_status VARCHAR2(20) := 'NOW';
        v_meta   VARCHAR2(32767) := NULL;
      BEGIN
        v_raw := :body_text;
        IF v_raw IS NULL OR DBMS_LOB.GETLENGTH(v_raw) = 0 THEN
          RAISE_APPLICATION_ERROR(-20050, 'Dá»¯ liá»‡u JSON gá»­i lÃªn bá»‹ rá»—ng');
        END IF;

        v_obj := JSON_OBJECT_T.parse(v_raw);

        IF v_obj.has('movie_id') AND NOT v_obj.get('movie_id').is_null THEN
          v_id := TO_NUMBER(TRIM(v_obj.get_String('movie_id')));
        END IF;

        v_title  := v_obj.get_String('title');
        v_genre  := v_obj.get_String('genre');
        v_dur    := TO_NUMBER(TRIM(v_obj.get_String('duration_min')));
        v_rating := v_obj.get_String('rating');

        IF v_obj.has('status') AND NOT v_obj.get('status').is_null THEN
          v_status := NVL(TRIM(v_obj.get_String('status')), 'NOW');
        END IF;

        IF v_obj.has('metadata') AND NOT v_obj.get('metadata').is_null THEN
          v_elem := v_obj.get('metadata');
          IF v_elem.is_string THEN
            v_meta := v_obj.get_String('metadata');
          ELSE
            v_meta := v_elem.stringify();
          END IF;
        END IF;

        IF v_id IS NOT NULL THEN
          UPDATE movie
             SET title        = v_title,
                 genre        = v_genre,
                 duration_min = v_dur,
                 rating       = v_rating,
                 status       = v_status
           WHERE movie_id = v_id;
          IF v_meta IS NOT NULL THEN
            UPDATE movie SET metadata = v_meta WHERE movie_id = v_id;
          END IF;

-- 1. Sá»­a láº¡i hÃ m FN_CALC_DISCOUNT (Ä‘Æ°a JSON_SERIALIZE vÃ o cÃ¢u SELECT SQL)
CREATE OR REPLACE FUNCTION fn_calc_discount(
  p_promo_code  IN VARCHAR2,
  p_subtotal    IN NUMBER,
  p_showtime_id IN NUMBER
) RETURN NUMBER IS
  v_type       promotion.discount_type%TYPE;
  v_val        promotion.discount_value%TYPE;
  v_max        promotion.max_discount%TYPE;
  v_min        promotion.min_amount%TYPE;
  v_from       promotion.valid_from%TYPE;
  v_to         promotion.valid_to%TYPE;
  v_max_uses   promotion.max_uses%TYPE;
  v_used       promotion.used_count%TYPE;
  v_active     promotion.is_active%TYPE;
  v_cond_txt   VARCHAR2(4000);
  v_start_time showtime.start_time%TYPE;
  v_show_day   VARCHAR2(10);
  v_obj        JSON_OBJECT_T;
  v_arr        JSON_ARRAY_T;
  v_day_ok     BOOLEAN;
  v_disc       NUMBER := 0;
BEGIN
  IF p_promo_code IS NULL OR TRIM(p_promo_code) = '' THEN
    RETURN 0;
  END IF;

  BEGIN
    SELECT discount_type, discount_value, max_discount, NVL(min_amount, 0),
           valid_from, valid_to, max_uses, used_count, is_active,
           JSON_SERIALIZE(conditions RETURNING VARCHAR2(4000))
      INTO v_type, v_val, v_max, v_min,
           v_from, v_to, v_max_uses, v_used, v_active, v_cond_txt
      FROM promotion
     WHERE code = UPPER(TRIM(p_promo_code));
  EXCEPTION
    WHEN NO_DATA_FOUND THEN
      RAISE_APPLICATION_ERROR(-20034, 'MÃ£ khuyáº¿n mÃ£i khÃ´ng tá»“n táº¡i');
  END;

  IF v_active <> 1 OR TRUNC(SYSDATE) NOT BETWEEN v_from AND v_to THEN
    RAISE_APPLICATION_ERROR(-20034, 'MÃ£ khuyáº¿n mÃ£i khÃ´ng tá»“n táº¡i hoáº·c Ä‘Ã£ háº¿t háº¡n');
  END IF;

  IF v_max_uses IS NOT NULL AND v_used >= v_max_uses THEN
    RAISE_APPLICATION_ERROR(-20031, 'MÃ£ khuyáº¿n mÃ£i Ä‘Ã£ háº¿t lÆ°á»£t dÃ¹ng');
  END IF;

  IF p_subtotal < v_min THEN
    RAISE_APPLICATION_ERROR(-20032, 'ChÆ°a Ä‘áº¡t giÃ¡ trá»‹ Ä‘Æ¡n tá»‘i thiá»ƒu');
  END IF;

  IF v_cond_txt IS NOT NULL THEN
    v_obj := JSON_OBJECT_T.parse(v_cond_txt);
    IF v_obj.has('days') AND NOT v_obj.get('days').is_null THEN
      v_arr := v_obj.get_Array('days');
      IF v_arr.get_size > 0 THEN
        SELECT start_time INTO v_start_time
          FROM showtime WHERE showtime_id = p_showtime_id;
        v_show_day := TO_CHAR(v_start_time, 'DY', 'NLS_DATE_LANGUAGE=AMERICAN');
        v_day_ok := FALSE;
        FOR i IN 0 .. v_arr.get_size - 1 LOOP
          IF UPPER(v_arr.get_String(i)) = v_show_day THEN
            v_day_ok := TRUE;
            EXIT;
          END IF;
        END LOOP;
        IF NOT v_day_ok THEN
          RAISE_APPLICATION_ERROR(-20033, 'MÃ£ khÃ´ng Ã¡p dá»¥ng cho ngÃ y chiáº¿u nÃ y');
        END IF;
      END IF;
    END IF;
  END IF;

  IF v_type = 'PERCENT' THEN
    v_disc := p_subtotal * v_val / 100;
  ELSE
    v_disc := v_val;
  END IF;

  IF v_max IS NOT NULL THEN
    v_disc := LEAST(v_disc, v_max);
  END IF;

  RETURN LEAST(ROUND(v_disc, -3), p_subtotal);
END fn_calc_discount;

  PROCEDURE sp_hold_seats(
    p_customer_id IN NUMBER, p_showtime_id IN NUMBER,
    p_seat_ids IN SYS.ODCINUMBERLIST, p_booking_id OUT NUMBER) IS
    c_hold_minutes CONSTANT NUMBER := 10;
    v_expire TIMESTAMP := SYSTIMESTAMP + NUMTODSINTERVAL(c_hold_minutes, 'MINUTE');
    v_start TIMESTAMP; v_status VARCHAR2(10);
  BEGIN
    SELECT start_time, status INTO v_start, v_status
      FROM showtime WHERE showtime_id = p_showtime_id;
    IF v_status <> 'OPEN' OR v_start <= SYSTIMESTAMP THEN
      RAISE_APPLICATION_ERROR(-20012, 'Suáº¥t chiáº¿u Ä‘Ã£ Ä‘Ã³ng bÃ¡n');
    END IF;

    IF p_seat_ids IS NULL OR p_seat_ids.COUNT NOT BETWEEN 1 AND 8 THEN
      RAISE_APPLICATION_ERROR(-20010, 'Má»—i Ä‘Æ¡n Ä‘áº·t tá»« 1 Ä‘áº¿n 8 gháº¿');
    END IF;

    SAVEPOINT sp_start;

    INSERT INTO booking(customer_id, showtime_id, status, expires_at)
    VALUES (p_customer_id, p_showtime_id, 'PENDING', v_expire)
    RETURNING booking_id INTO p_booking_id;

    FOR r IN (SELECT column_value AS seat_id FROM TABLE(p_seat_ids) ORDER BY 1) LOOP
      UPDATE showtime_seat
         SET status = 'HELD', held_by = p_customer_id,
             hold_expires_at = v_expire, booking_id = p_booking_id
       WHERE showtime_id = p_showtime_id AND seat_id = r.seat_id

              OR (status = 'HELD' AND hold_expires_at < SYSTIMESTAMP));
      IF SQL%ROWCOUNT = 0 THEN
        ROLLBACK TO sp_start;
        RAISE_APPLICATION_ERROR(-20011,
          'Gháº¿ ' || r.seat_id || ' Ä‘Ã£ cÃ³ ngÆ°á»i giá»¯ hoáº·c Ä‘Ã£ bÃ¡n');
      END IF;
    END LOOP;
    COMMIT;
  END sp_hold_seats;

  PROCEDURE sp_confirm_booking(
    p_booking_id IN NUMBER, p_promo_code IN VARCHAR2,
    p_method IN VARCHAR2, p_combos IN VARCHAR2 DEFAULT NULL) IS
    v_status      VARCHAR2(10);
    v_expires_at  TIMESTAMP;
    v_showtime_id NUMBER;
    v_total NUMBER; v_cnt NUMBER;
    v_subtotal NUMBER; v_discount NUMBER;

  BEGIN
    SELECT status, expires_at, showtime_id
      INTO v_status, v_expires_at, v_showtime_id
      FROM booking WHERE booking_id = p_booking_id FOR UPDATE;
    IF v_status <> 'PENDING' OR v_expires_at < SYSTIMESTAMP THEN
      RAISE_APPLICATION_ERROR(-20020, 'ÄÆ¡n khÃ´ng cÃ²n hiá»‡u lá»±c');
    END IF;

    INSERT INTO ticket(booking_id, showtime_id, seat_id, price)
    SELECT booking_id, showtime_id, seat_id, fn_calc_ticket_price(showtime_id, seat_id)
      FROM showtime_seat
     WHERE booking_id = p_booking_id AND status = 'HELD'
       AND hold_expires_at >= SYSTIMESTAMP;
    v_cnt := SQL%ROWCOUNT;
    IF v_cnt = 0 OR v_cnt <> v_total THEN
      RAISE_APPLICATION_ERROR(-20021, 'CÃ³ gháº¿ Ä‘Ã£ háº¿t háº¡n giá»¯ chá»—');
    END IF;

    IF p_combos IS NOT NULL AND TRIM(p_combos) NOT IN ('', '{}', 'null') THEN

      v_keys := v_obj.get_keys;
      FOR i IN 1 .. v_keys.COUNT LOOP
        v_cid := TO_NUMBER(v_keys(i));
        v_qty := v_obj.get_Number(v_keys(i));
        IF v_qty > 0 THEN

          INSERT INTO booking_combo(booking_id, combo_id, qty, unit_price)
          VALUES (p_booking_id, v_cid, v_qty, v_cprice);
          v_combo_amt := v_combo_amt + (v_qty * v_cprice);
        END IF;
      END LOOP;
    END IF;

    SELECT SUM(price) INTO v_subtotal FROM ticket WHERE booking_id = p_booking_id;
    v_discount := fn_calc_discount(p_promo_code, v_subtotal, v_showtime_id);

    IF v_discount > 0 THEN
      UPDATE promotion SET used_count = used_count + 1
       WHERE code = UPPER(p_promo_code)

      IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(-20031, 'MÃ£ khuyáº¿n mÃ£i Ä‘Ã£ háº¿t lÆ°á»£t dÃ¹ng');
      END IF;
    END IF;

     WHERE booking_id = p_booking_id;
    UPDATE booking SET subtotal = v_subtotal, discount_amount = v_discount,
           combo_amount = v_combo_amt,
           final_amount = v_subtotal - v_discount + v_combo_amt,
           promo_code = UPPER(p_promo_code),
           status = 'PAID', paid_at = SYSTIMESTAMP
     WHERE booking_id = p_booking_id;
    INSERT INTO payment(booking_id, method, amount, status, paid_at)

  PROCEDURE sp_daily_revenue_report(p_date IN DATE DEFAULT TRUNC(SYSDATE)) IS
    CURSOR c_revenue IS
      SELECT m.title, COUNT(*) AS tickets_sold, SUM(t.price) AS revenue
        FROM ticket t
        JOIN booking  b  ON b.booking_id = t.booking_id AND b.status = 'PAID'
        JOIN showtime st ON st.showtime_id = t.showtime_id
        JOIN movie    m  ON m.movie_id = st.movie_id
       WHERE TRUNC(b.paid_at) = p_date
       GROUP BY m.title
       ORDER BY SUM(t.price) DESC;
    v_title VARCHAR2(200); v_tickets_sold NUMBER; v_revenue NUMBER;
    v_total_rev NUMBER := 0; v_total_tkt NUMBER := 0;
  BEGIN
    DBMS_OUTPUT.PUT_LINE('BAO CAO DOANH THU NGAY ' || TO_CHAR(p_date, 'DD/MM/YYYY'));
    OPEN c_revenue;
    LOOP
      FETCH c_revenue INTO v_title, v_tickets_sold, v_revenue;
      EXIT WHEN c_revenue%NOTFOUND;
      DBMS_OUTPUT.PUT_LINE(RPAD(v_title, 35) || LPAD(v_tickets_sold, 10) || LPAD(TO_CHAR(v_revenue, '999,999,999'), 15));
      v_total_rev := v_total_rev + v_revenue;
      v_total_tkt := v_total_tkt + v_tickets_sold;
    END LOOP;
    CLOSE c_revenue;
    DBMS_OUTPUT.PUT_LINE(RPAD('TONG CONG', 35) || LPAD(v_total_tkt, 10) || LPAD(TO_CHAR(v_total_rev, '999,999,999'), 15));
  EXCEPTION
    WHEN OTHERS THEN
      IF c_revenue%ISOPEN THEN CLOSE c_revenue; END IF;
      RAISE;
  END sp_daily_revenue_report;

  PROCEDURE sp_cancel_booking(p_booking_id IN NUMBER, p_customer_id IN NUMBER) IS
    v_bk_customer_id NUMBER; v_status VARCHAR2(10); v_showtime_id NUMBER;
    v_promo_code VARCHAR2(30); v_final_amount NUMBER; v_start TIMESTAMP;
  BEGIN

    IF v_status = 'PAID' THEN

        RAISE_APPLICATION_ERROR(-20042, 'Chá»‰ huá»· vÃ© Ä‘Ã£ thanh toÃ¡n trÆ°á»›c giá» chiáº¿u Ã­t nháº¥t 2 giá»');
      END IF;

    COMMIT;
  EXCEPTION
    WHEN OTHERS THEN
      ROLLBACK;
      RAISE;
  END sp_cancel_booking;

  FUNCTION fn_check_room_overlap(
    p_room_id NUMBER, p_start TIMESTAMP, p_end TIMESTAMP,
    p_exclude_id NUMBER DEFAULT NULL) RETURN NUMBER IS
    v_cnt NUMBER;
  BEGIN
    SELECT COUNT(*) INTO v_cnt
      FROM showtime s
     WHERE s.room_id = p_room_id
       AND s.status <> 'CANCELLED'
       AND (p_exclude_id IS NULL OR s.showtime_id <> p_exclude_id)
       AND s.start_time < p_end
       AND s.end_time   > p_start;
    RETURN v_cnt;
  END fn_check_room_overlap;

  showtime_id     NUMBER NOT NULL REFERENCES showtime(showtime_id),
  promo_code      VARCHAR2(30),
  subtotal        NUMBER,
  discount_amount NUMBER,
  final_amount    NUMBER,
  status          VARCHAR2(10) DEFAULT 'PENDING' NOT NULL
                  CHECK (status IN ('PENDING','PAID','CANCELLED','EXPIRED')),
  created_at      TIMESTAMP DEFAULT SYSTIMESTAMP NOT NULL,
  expires_at      TIMESTAMP,
  paid_at         TIMESTAMP
);

-- 9) SHOWTIME_SEAT â€” báº£ng trung tÃ¢m chá»‘ng bÃ¡n trÃ¹ng gháº¿ (má»¥c 3.1 bÃ¡o cÃ¡o)
--    ChÃº Ã½: booking_id KHÃ”NG Ä‘áº·t FOREIGN KEY tá»›i BOOKING â€” Ä‘á»ƒ trÃ¡nh vÃ²ng
--    phá»¥ thuá»™c (BOOKING cÅ©ng tham chiáº¿u SHOWTIME), quan há»‡ nÃ y Ä‘Æ°á»£c kiá»ƒm
--    soÃ¡t hoÃ n toÃ n trong PL/SQL cá»§a pkg_booking.
CREATE TABLE showtime_seat (
  showtime_id     NUMBER NOT NULL REFERENCES showtime(showtime_id),
  seat_id         NUMBER NOT NULL REFERENCES seat(seat_id),
  status          VARCHAR2(10) DEFAULT 'AVAILABLE' NOT NULL
                  CHECK (status IN ('AVAILABLE','HELD','SOLD')),
  held_by         NUMBER REFERENCES customer(customer_id),
  hold_expires_at TIMESTAMP,
  booking_id      NUMBER,
  CONSTRAINT pk_showtime_seat PRIMARY KEY (showtime_id, seat_id),
  CONSTRAINT ck_hold CHECK (status <> 'HELD'
                            OR (held_by IS NOT NULL AND hold_expires_at IS NOT NULL))
);

-- 10) TICKET â€” phá»¥ thuá»™c BOOKING vÃ  SHOWTIME_SEAT (khoÃ¡ ngoáº¡i tá»• há»£p)
CREATE TABLE ticket (
  ticket_id   NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  booking_id  NUMBER NOT NULL REFERENCES booking(booking_id),
  showtime_id NUMBER NOT NULL,
  seat_id     NUMBER NOT NULL,
  price       NUMBER(10) NOT NULL CHECK (price >= 0),
  CONSTRAINT uq_ticket_seat UNIQUE (showtime_id, seat_id),
  CONSTRAINT fk_ticket_ss FOREIGN KEY (showtime_id, seat_id)
    REFERENCES showtime_seat(showtime_id, seat_id)
);

-- 11) PAYMENT â€” phá»¥ thuá»™c BOOKING
CREATE TABLE payment (
  payment_id NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  booking_id NUMBER NOT NULL REFERENCES booking(booking_id),
  method     VARCHAR2(20) NOT NULL,
  amount     NUMBER NOT NULL,
  status     VARCHAR2(20) NOT NULL CHECK (status IN ('SUCCESS','FAILED','REFUNDED')),
  paid_at    TIMESTAMP DEFAULT SYSTIMESTAMP
);

  old_value  VARCHAR2(50), new_value VARCHAR2(50),
  changed_by VARCHAR2(128) DEFAULT SYS_CONTEXT('USERENV', 'SESSION_USER'),
  changed_at TIMESTAMP DEFAULT SYSTIMESTAMP

-- 3.2. HÃ m tÃ­nh giÃ¡ vÃ©
CREATE OR REPLACE FUNCTION fn_calc_ticket_price(
  p_showtime_id NUMBER, p_seat_id NUMBER) RETURN NUMBER IS
  -- Khai bÃ¡o kiá»ƒu dá»¯ liá»‡u cá»¥ thá»ƒ cho tá»«ng biáº¿n (khÃ´ng dÃ¹ng %TYPE)
  v_base  NUMBER;
  v_start TIMESTAMP;
  v_type  VARCHAR2(10);
  v_price NUMBER;
BEGIN
  -- BÆ°á»›c 1: láº¥y giÃ¡ cÆ¡ sá»Ÿ cá»§a suáº¥t chiáº¿u vÃ  loáº¡i gháº¿
  SELECT st.base_price, st.start_time, s.seat_type
    INTO v_base, v_start, v_type
    FROM showtime st, seat s
   WHERE st.showtime_id = p_showtime_id AND s.seat_id = p_seat_id;

  -- BÆ°á»›c 2: nhÃ¢n há»‡ sá»‘ theo loáº¡i gháº¿ (gháº¿ thÆ°á»ng x1, VIP x1.25, COUPLE x2)
  v_price := v_base * CASE v_type WHEN 'VIP' THEN 1.25
                                  WHEN 'COUPLE' THEN 2.0 ELSE 1 END;

  -- BÆ°á»›c 3: cá»™ng thÃªm phá»¥ thu náº¿u suáº¥t chiáº¿u rÆ¡i vÃ o thá»© Báº£y/Chá»§ Nháº­t
  IF TO_CHAR(v_start, 'DY', 'NLS_DATE_LANGUAGE=ENGLISH') IN ('SAT','SUN') THEN
    v_price := v_price * 1.15;          -- phá»¥ thu cuá»‘i tuáº§n 15%
  END IF;

  -- BÆ°á»›c 4: giáº£m giÃ¡ náº¿u lÃ  suáº¥t chiáº¿u buá»•i sÃ¡ng sá»›m (trÆ°á»›c 12 giá» trÆ°a)
  IF EXTRACT(HOUR FROM v_start) < 12 THEN
    v_price := v_price * 0.8;           -- suáº¥t sá»›m giáº£m 20%
  END IF;

  RETURN ROUND(v_price, -3);            -- lÃ m trÃ²n Ä‘áº¿n nghÃ¬n Ä‘á»“ng
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    RAISE_APPLICATION_ERROR(-20001, 'Suáº¥t chiáº¿u hoáº·c gháº¿ khÃ´ng tá»“n táº¡i');
END;
/

-- 3.3. HÃ m tÃ­nh khuyáº¿n mÃ£i
CREATE OR REPLACE FUNCTION fn_calc_discount(
  p_code VARCHAR2, p_subtotal NUMBER, p_showtime_id NUMBER) RETURN NUMBER IS
  -- Khai bÃ¡o riÃªng tá»«ng biáº¿n vá»›i kiá»ƒu dá»¯ liá»‡u cá»¥ thá»ƒ (khÃ´ng gá»™p chung vÃ o %ROWTYPE)
  v_valid_from     DATE;
  v_valid_to       DATE;
  v_max_uses       NUMBER;
  v_used_count     NUMBER;
  v_min_amount     NUMBER;
  v_discount_type  VARCHAR2(10);
  v_discount_value NUMBER;
  v_max_discount   NUMBER;
  v_conditions     VARCHAR2(4000);  -- cá»™t JSON Ä‘Æ°á»£c lÆ°u dÆ°á»›i dáº¡ng chuá»—i kÃ½ tá»±
  v_dow    VARCHAR2(3);     -- thá»© trong tuáº§n cá»§a suáº¥t chiáº¿u, vÃ­ dá»¥ 'TUE'
  v_days   VARCHAR2(200);   -- ná»™i dung JSON "days" dáº¡ng chuá»—i, vÃ­ dá»¥ ["TUE","WED"]
  v_disc   NUMBER;
BEGIN
  -- KhÃ´ng nháº­p mÃ£ thÃ¬ khÃ´ng giáº£m, khá»i cáº§n kiá»ƒm tra gÃ¬ thÃªm
  IF p_code IS NULL THEN RETURN 0; END IF;

  -- BÆ°á»›c 1: láº¥y tá»«ng cá»™t cáº§n dÃ¹ng cá»§a mÃ£ khuyáº¿n mÃ£i ra tá»«ng biáº¿n riÃªng
  -- (náº¿u khÃ´ng tÃ¬m tháº¥y mÃ£ sáº½ rÆ¡i vÃ o NO_DATA_FOUND)
  SELECT valid_from, valid_to, max_uses, used_count, min_amount,
         discount_type, discount_value, max_discount, conditions
    INTO v_valid_from, v_valid_to, v_max_uses, v_used_count, v_min_amount,
         v_discount_type, v_discount_value, v_max_discount, v_conditions
    FROM promotion WHERE code = UPPER(p_code) AND is_active = 1;

  -- BÆ°á»›c 2: kiá»ƒm tra cÃ¡c Ä‘iá»u kiá»‡n cÆ¡ báº£n cá»§a mÃ£
  IF SYSDATE NOT BETWEEN v_valid_from AND v_valid_to THEN
    RAISE_APPLICATION_ERROR(-20030, 'MÃ£ khuyáº¿n mÃ£i Ä‘Ã£ háº¿t háº¡n');
  ELSIF v_max_uses IS NOT NULL AND v_used_count >= v_max_uses THEN
    RAISE_APPLICATION_ERROR(-20031, 'MÃ£ khuyáº¿n mÃ£i Ä‘Ã£ háº¿t lÆ°á»£t dÃ¹ng');
  ELSIF p_subtotal < NVL(v_min_amount, 0) THEN
    RAISE_APPLICATION_ERROR(-20032, 'ChÆ°a Ä‘áº¡t giÃ¡ trá»‹ Ä‘Æ¡n tá»‘i thiá»ƒu');
  END IF;

  -- BÆ°á»›c 3: kiá»ƒm tra Ä‘iá»u kiá»‡n "chá»‰ Ã¡p dá»¥ng má»™t sá»‘ ngÃ y trong tuáº§n"
  -- (Ä‘iá»u kiá»‡n nÃ y lÆ°u trong cá»™t JSON conditions, vÃ­ dá»¥ {"days":["TUE","WED"]})
  IF v_conditions IS NOT NULL AND JSON_EXISTS(v_conditions, '$.days') THEN
    SELECT TO_CHAR(start_time, 'DY', 'NLS_DATE_LANGUAGE=ENGLISH')
      INTO v_dow FROM showtime WHERE showtime_id = p_showtime_id;

    -- Chuyá»ƒn cá»¥c JSON thÃ nh chuá»—i kÃ½ tá»± rá»“i tÃ¬m xem cÃ³ chá»©a Ä‘Ãºng
    -- mÃ£ 3 kÃ½ tá»± cá»§a ngÃ y chiáº¿u hay khÃ´ng (vÃ­ dá»¥ tÃ¬m "TUE" trong chuá»—i)
    v_days := JSON_SERIALIZE(v_conditions RETURNING VARCHAR2(200));
    IF INSTR(v_days, '"' || v_dow || '"') = 0 THEN
      RAISE_APPLICATION_ERROR(-20033, 'MÃ£ khÃ´ng Ã¡p dá»¥ng cho ngÃ y chiáº¿u nÃ y');
    END IF;
  END IF;

  -- BÆ°á»›c 4: tÃ­nh sá»‘ tiá»n giáº£m theo loáº¡i khuyáº¿n mÃ£i, cháº·n tráº§n max_discount
  v_disc := CASE v_discount_type WHEN 'PERCENT'
              THEN p_subtotal * v_discount_value / 100
              ELSE v_discount_value END;
  IF v_max_discount IS NOT NULL THEN v_disc := LEAST(v_disc, v_max_discount); END IF;
  RETURN LEAST(ROUND(v_disc, -3), p_subtotal);
EXCEPTION
  WHEN NO_DATA_FOUND THEN
    RAISE_APPLICATION_ERROR(-20034, 'MÃ£ khuyáº¿n mÃ£i khÃ´ng tá»“n táº¡i');
END;
/

-- ---------- BODY ----------
CREATE OR REPLACE PACKAGE BODY pkg_booking IS

  -- 3.4. Giá»¯ chá»— táº¡m thá»i
  PROCEDURE sp_hold_seats(
    p_customer_id IN NUMBER, p_showtime_id IN NUMBER,
    p_seat_ids IN SYS.ODCINUMBERLIST, p_booking_id OUT NUMBER) IS
    c_hold_minutes CONSTANT NUMBER := 10;
    v_expire TIMESTAMP := SYSTIMESTAMP + NUMTODSINTERVAL(c_hold_minutes, 'MINUTE');
    v_start TIMESTAMP; v_status VARCHAR2(10);
  BEGIN
    -- BÆ°á»›c 1: cháº·n Ä‘áº·t suáº¥t Ä‘Ã£ Ä‘Ã³ng bÃ¡n hoáº·c Ä‘Ã£ báº¯t Ä‘áº§u chiáº¿u
    SELECT start_time, status INTO v_start, v_status
      FROM showtime WHERE showtime_id = p_showtime_id;
    IF v_status <> 'OPEN' OR v_start <= SYSTIMESTAMP THEN
      RAISE_APPLICATION_ERROR(-20012, 'Suáº¥t chiáº¿u Ä‘Ã£ Ä‘Ã³ng bÃ¡n');
    END IF;

    -- BÆ°á»›c 2: kiá»ƒm tra sá»‘ lÆ°á»£ng gháº¿ há»£p lá»‡ (1 Ä‘áº¿n 8 gháº¿ má»™t Ä‘Æ¡n)
    IF p_seat_ids IS NULL OR p_seat_ids.COUNT NOT BETWEEN 1 AND 8 THEN
      RAISE_APPLICATION_ERROR(-20010, 'Má»—i Ä‘Æ¡n Ä‘áº·t tá»« 1 Ä‘áº¿n 8 gháº¿');
    END IF;

    -- BÆ°á»›c 3: Ä‘Ã¡nh dáº¥u Ä‘iá»ƒm má»‘c Ä‘á»ƒ cÃ³ thá»ƒ hoÃ n tÃ¡c (ROLLBACK TO) náº¿u má»™t
    -- gháº¿ nÃ o Ä‘Ã³ bá»‹ trÃ¹ng á»Ÿ giá»¯a chá»«ng, thay vÃ¬ pháº£i huá»· luÃ´n toÃ n bá»™ giao dá»‹ch
    SAVEPOINT sp_start;

    -- BÆ°á»›c 4: táº¡o má»™t Ä‘Æ¡n Ä‘áº·t vÃ© má»›i á»Ÿ tráº¡ng thÃ¡i PENDING (Ä‘ang chá» thanh toÃ¡n)
    INSERT INTO booking(customer_id, showtime_id, status, expires_at)
    VALUES (p_customer_id, p_showtime_id, 'PENDING', v_expire)
    RETURNING booking_id INTO p_booking_id;

    -- BÆ°á»›c 5: láº§n lÆ°á»£t giá»¯ tá»«ng gháº¿ trong danh sÃ¡ch p_seat_ids.
    -- TABLE(p_seat_ids) biáº¿n táº­p há»£p sá»‘ thÃ nh má»™t "báº£ng áº£o" Ä‘á»ƒ duyá»‡t báº±ng FOR,
    -- giá»‘ng nhÆ° duyá»‡t qua tá»«ng pháº§n tá»­ cá»§a má»™t máº£ng.
    FOR r IN (SELECT column_value AS seat_id FROM TABLE(p_seat_ids) ORDER BY 1) LOOP
      -- Chá»‰ UPDATE khi gháº¿ Ä‘ang trá»‘ng, hoáº·c Ä‘ang bá»‹ giá»¯ nhÆ°ng Ä‘Ã£ quÃ¡ háº¡n
      UPDATE showtime_seat
         SET status = 'HELD', held_by = p_customer_id,
             hold_expires_at = v_expire, booking_id = p_booking_id
       WHERE showtime_id = p_showtime_id AND seat_id = r.seat_id
         AND (status = 'AVAILABLE'
              OR (status = 'HELD' AND hold_expires_at < SYSTIMESTAMP));
      -- KhÃ´ng cÃ³ dÃ²ng nÃ o Ä‘Æ°á»£c cáº­p nháº­t nghÄ©a lÃ  gháº¿ Ä‘Ã£ bá»‹ ngÆ°á»i khÃ¡c giá»¯/bÃ¡n
      IF SQL%ROWCOUNT = 0 THEN
        ROLLBACK TO sp_start;   -- tráº£ toÃ n bá»™ Ä‘Æ¡n vá» tráº¡ng thÃ¡i ban Ä‘áº§u
        RAISE_APPLICATION_ERROR(-20011,
          'Gháº¿ ' || r.seat_id || ' Ä‘Ã£ cÃ³ ngÆ°á»i giá»¯ hoáº·c Ä‘Ã£ bÃ¡n');
      END IF;
    END LOOP;
    COMMIT;   -- má»i gháº¿ Ä‘á»u giá»¯ thÃ nh cÃ´ng má»›i COMMIT
  END sp_hold_seats;

  -- 3.5. XÃ¡c nháº­n thanh toÃ¡n
  PROCEDURE sp_confirm_booking(
    p_booking_id IN NUMBER, p_promo_code IN VARCHAR2, p_method IN VARCHAR2, p_combos     IN VARCHAR2 DEFAULT NULL) IS
    -- Khai bÃ¡o riÃªng tá»«ng cá»™t cáº§n dÃ¹ng cá»§a Ä‘Æ¡n Ä‘áº·t vÃ© (khÃ´ng dÃ¹ng %ROWTYPE)
    v_status      VARCHAR2(10);
    v_expires_at  TIMESTAMP;
    v_showtime_id NUMBER;
    v_total NUMBER; v_cnt NUMBER;
    v_subtotal NUMBER; v_discount NUMBER;
  BEGIN
    -- BÆ°á»›c 1: khoÃ¡ dÃ²ng Ä‘Æ¡n Ä‘áº·t láº¡i (FOR UPDATE) Ä‘á»ƒ khÃ´ng ai báº¥m thanh toÃ¡n
    -- trÃ¹ng láº·p cho cÃ¹ng má»™t Ä‘Æ¡n, rá»“i kiá»ƒm tra Ä‘Æ¡n cÃ²n hiá»‡u lá»±c khÃ´ng
    SELECT status, expires_at, showtime_id
      INTO v_status, v_expires_at, v_showtime_id
      FROM booking WHERE booking_id = p_booking_id FOR UPDATE;
    IF v_status <> 'PENDING' OR v_expires_at < SYSTIMESTAMP THEN
      RAISE_APPLICATION_ERROR(-20020, 'ÄÆ¡n khÃ´ng cÃ²n hiá»‡u lá»±c');
    END IF;

    -- BÆ°á»›c 2: táº¡o vÃ© (ticket) cho tá»«ng gháº¿ Ä‘ang HELD vÃ  cÃ²n háº¡n giá»¯ chá»—,
    -- giÃ¡ má»—i vÃ© Ä‘Æ°á»£c tÃ­nh ngay báº±ng hÃ m fn_calc_ticket_price
    SELECT COUNT(*) INTO v_total FROM showtime_seat WHERE booking_id = p_booking_id;

    INSERT INTO ticket(booking_id, showtime_id, seat_id, price)
    SELECT booking_id, showtime_id, seat_id, fn_calc_ticket_price(showtime_id, seat_id)
      FROM showtime_seat
     WHERE booking_id = p_booking_id AND status = 'HELD'
       AND hold_expires_at >= SYSTIMESTAMP;
    v_cnt := SQL%ROWCOUNT;
    -- Náº¿u sá»‘ vÃ© táº¡o Ä‘Æ°á»£c Ã­t hÆ¡n sá»‘ gháº¿ Ä‘Ã£ giá»¯, nghÄ©a lÃ  cÃ³ gháº¿ bá»‹ háº¿t háº¡n giá»¯a chá»«ng
    IF v_cnt = 0 OR v_cnt <> v_total THEN
      RAISE_APPLICATION_ERROR(-20021, 'CÃ³ gháº¿ Ä‘Ã£ háº¿t háº¡n giá»¯ chá»—');
    END IF;

    -- BÆ°á»›c 3: tÃ­nh tá»•ng tiá»n vÃ  sá»‘ tiá»n Ä‘Æ°á»£c giáº£m (náº¿u cÃ³ mÃ£ khuyáº¿n mÃ£i)
    SELECT SUM(price) INTO v_subtotal FROM ticket WHERE booking_id = p_booking_id;
    v_discount := fn_calc_discount(p_promo_code, v_subtotal, v_showtime_id);

    -- BÆ°á»›c 4: náº¿u cÃ³ giáº£m giÃ¡ thÃ¬ tÄƒng used_count cá»§a mÃ£ lÃªn 1,
    -- nhÆ°ng chá»‰ tÄƒng khi used_count váº«n cÃ²n dÆ°á»›i max_uses (trÃ¡nh vÆ°á»£t quÃ¡)
    IF v_discount > 0 THEN
      UPDATE promotion SET used_count = used_count + 1
       WHERE code = UPPER(p_promo_code)
         AND (max_uses IS NULL OR used_count < max_uses);
      IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(-20031, 'MÃ£ khuyáº¿n mÃ£i Ä‘Ã£ háº¿t lÆ°á»£t dÃ¹ng');
      END IF;
    END IF;

    -- BÆ°á»›c 5: chuyá»ƒn gháº¿ sang SOLD, chuyá»ƒn Ä‘Æ¡n sang PAID, ghi nháº­n thanh toÃ¡n.
    -- ToÃ n bá»™ 5 bÆ°á»›c náº±m trong má»™t giao dá»‹ch duy nháº¥t nhá» chá»‰ COMMIT á»Ÿ cuá»‘i cÃ¹ng.
    UPDATE showtime_seat SET status = 'SOLD', held_by = NULL, hold_expires_at = NULL
     WHERE booking_id = p_booking_id;
    UPDATE booking SET subtotal = v_subtotal, discount_amount = v_discount,
           final_amount = v_subtotal - v_discount, promo_code = UPPER(p_promo_code),
           status = 'PAID', paid_at = SYSTIMESTAMP
     WHERE booking_id = p_booking_id;
    INSERT INTO payment(booking_id, method, amount, status, paid_at)
    VALUES (p_booking_id, p_method, v_subtotal - v_discount, 'SUCCESS', SYSTIMESTAMP);
    COMMIT;
  EXCEPTION
    WHEN OTHERS THEN
      ROLLBACK;   -- cÃ³ lá»—i á»Ÿ báº¥t ká»³ bÆ°á»›c nÃ o thÃ¬ huá»· háº¿t, khÃ´ng Ä‘á»ƒ dá»¯ liá»‡u ná»­a vá»i
      RAISE;
  END sp_confirm_booking;

  -- 3.6. Giáº£i phÃ³ng gháº¿ giá»¯ quÃ¡ háº¡n (Ä‘Æ°á»£c DBMS_SCHEDULER gá»i má»—i phÃºt)
  PROCEDURE sp_release_expired_holds IS
  BEGIN
    UPDATE showtime_seat
       SET status = 'AVAILABLE', held_by = NULL, hold_expires_at = NULL, booking_id = NULL
     WHERE status = 'HELD' AND hold_expires_at < SYSTIMESTAMP;
    UPDATE booking SET status = 'EXPIRED'
     WHERE status = 'PENDING' AND expires_at < SYSTIMESTAMP;
    COMMIT;
  END sp_release_expired_holds;

  -- 3.7. BÃ¡o cÃ¡o doanh thu cuá»‘i ngÃ y â€” Con trá» tÆ°á»ng minh (Explicit Cursor)
  PROCEDURE sp_daily_revenue_report(p_date IN DATE DEFAULT TRUNC(SYSDATE)) IS
    -- Khai bÃ¡o con trá» tÆ°á»ng minh
    CURSOR c_revenue IS
      SELECT m.title, COUNT(*) AS tickets_sold, SUM(t.price) AS revenue
        FROM ticket t
        JOIN booking  b  ON b.booking_id = t.booking_id AND b.status = 'PAID'
        JOIN showtime st ON st.showtime_id = t.showtime_id
        JOIN movie    m  ON m.movie_id = st.movie_id
       WHERE TRUNC(b.paid_at) = p_date
       GROUP BY m.title
       ORDER BY SUM(t.price) DESC;

    -- Khai bÃ¡o riÃªng tá»«ng biáº¿n tÆ°Æ¡ng á»©ng vá»›i tá»«ng cá»™t cursor tráº£ vá» (khÃ´ng dÃ¹ng %ROWTYPE)
    v_title        VARCHAR2(200);
    v_tickets_sold NUMBER;
    v_revenue      NUMBER;
    v_total_rev NUMBER := 0;
    v_total_tkt NUMBER := 0;
  BEGIN
    DBMS_OUTPUT.PUT_LINE('BAO CAO DOANH THU NGAY ' || TO_CHAR(p_date, 'DD/MM/YYYY'));
    DBMS_OUTPUT.PUT_LINE(RPAD('Phim', 35) || LPAD('So ve', 10) || LPAD('Doanh thu', 15));

    OPEN c_revenue;                      -- má»Ÿ con trá»
    LOOP
      FETCH c_revenue INTO v_title, v_tickets_sold, v_revenue;  -- láº¥y tá»«ng dÃ²ng
      EXIT WHEN c_revenue%NOTFOUND;      -- háº¿t dÃ²ng thÃ¬ dá»«ng

      DBMS_OUTPUT.PUT_LINE(RPAD(v_title, 35)
                            || LPAD(v_tickets_sold, 10)
                            || LPAD(TO_CHAR(v_revenue, '999,999,999'), 15));
      v_total_rev := v_total_rev + v_revenue;
      v_total_tkt := v_total_tkt + v_tickets_sold;
    END LOOP;
    CLOSE c_revenue;                     -- Ä‘Ã³ng con trá»

    DBMS_OUTPUT.PUT_LINE(RPAD('TONG CONG', 35) || LPAD(v_total_tkt, 10)
                          || LPAD(TO_CHAR(v_total_rev, '999,999,999'), 15));
  EXCEPTION
    WHEN OTHERS THEN
      IF c_revenue%ISOPEN THEN CLOSE c_revenue; END IF;  -- trÃ¡nh rÃ² rá»‰ con trá» khi lá»—i
      RAISE;
  END sp_daily_revenue_report;

  -- 3.11.b. Huá»· Ä‘Æ¡n vÃ  hoÃ n vÃ©
  PROCEDURE sp_cancel_booking(p_booking_id IN NUMBER, p_customer_id IN NUMBER) IS
    -- Khai bÃ¡o riÃªng tá»«ng cá»™t cáº§n dÃ¹ng cá»§a Ä‘Æ¡n Ä‘áº·t vÃ© (khÃ´ng dÃ¹ng %ROWTYPE/%TYPE)
    v_bk_customer_id NUMBER;
    v_status         VARCHAR2(10);
    v_showtime_id    NUMBER;
    v_promo_code     VARCHAR2(30);
    v_final_amount   NUMBER;
    v_start          TIMESTAMP;
  BEGIN
    -- BÆ°á»›c 1: khoÃ¡ dÃ²ng Ä‘Æ¡n, kiá»ƒm tra Ä‘Ãºng chá»§ Ä‘Æ¡n vÃ  Ä‘Ãºng tráº¡ng thÃ¡i Ä‘Æ°á»£c phÃ©p huá»·
    SELECT customer_id, status, showtime_id, promo_code, final_amount
      INTO v_bk_customer_id, v_status, v_showtime_id, v_promo_code, v_final_amount
      FROM booking WHERE booking_id = p_booking_id FOR UPDATE;
    IF v_bk_customer_id <> p_customer_id THEN
      RAISE_APPLICATION_ERROR(-20040, 'Khong co quyen huy don nay');
    ELSIF v_status NOT IN ('PENDING', 'PAID') THEN
      RAISE_APPLICATION_ERROR(-20041, 'Don khong the huy o trang thai hien tai');
    END IF;

    -- BÆ°á»›c 2: náº¿u Ä‘Æ¡n Ä‘Ã£ thanh toÃ¡n (PAID) thÃ¬ pháº£i hoÃ n tiá»n vÃ  xoÃ¡ vÃ©,
    -- nhÆ°ng chá»‰ Ä‘Æ°á»£c huá»· khi cÃ²n Ã­t nháº¥t 2 giá» trÆ°á»›c giá» chiáº¿u
    IF v_status = 'PAID' THEN
      SELECT start_time INTO v_start FROM showtime WHERE showtime_id = v_showtime_id;
      IF v_start < SYSTIMESTAMP + INTERVAL '2' HOUR THEN
        RAISE_APPLICATION_ERROR(-20042, 'Chi huy ve da thanh toan truoc gio chieu it nhat 2 gio');
      END IF;
      DELETE FROM ticket WHERE booking_id = p_booking_id;   -- xoÃ¡ vÃ© trÆ°á»›c
      INSERT INTO payment(booking_id, method, amount, status, paid_at)
      VALUES (p_booking_id, 'REFUND', -v_final_amount, 'REFUNDED', SYSTIMESTAMP);
      IF v_promo_code IS NOT NULL THEN
        UPDATE promotion SET used_count = GREATEST(used_count - 1, 0)
         WHERE code = v_promo_code;
      END IF;
    END IF;

    -- BÆ°á»›c 3: tráº£ gháº¿ vá» trá»‘ng vÃ  chuyá»ƒn Ä‘Æ¡n sang CANCELLED
    -- (náº¿u Ä‘Æ¡n Ä‘ang PENDING thÃ¬ chá»‰ cáº§n lÃ m Ä‘Ãºng bÆ°á»›c nÃ y, bá» qua bÆ°á»›c 2)
    UPDATE showtime_seat
       SET status = 'AVAILABLE', held_by = NULL, hold_expires_at = NULL, booking_id = NULL
     WHERE booking_id = p_booking_id;
    UPDATE booking SET status = 'CANCELLED' WHERE booking_id = p_booking_id;
    COMMIT;
  EXCEPTION
    WHEN OTHERS THEN
      ROLLBACK;
      RAISE;
  END sp_cancel_booking;

  -- 3.11.d. Chá»‘ng chá»“ng giá» chiáº¿u trong cÃ¹ng phÃ²ng
  FUNCTION fn_check_room_overlap(
    p_room_id NUMBER, p_start TIMESTAMP, p_end TIMESTAMP,
    p_exclude_id NUMBER DEFAULT NULL) RETURN NUMBER IS
    v_cnt NUMBER;
  BEGIN
    SELECT COUNT(*) INTO v_cnt
      FROM showtime s
     WHERE s.room_id = p_room_id
       AND s.status <> 'CANCELLED'
       AND (p_exclude_id IS NULL OR s.showtime_id <> p_exclude_id)
       AND s.start_time < p_end      -- suáº¥t má»›i báº¯t Ä‘áº§u trÆ°á»›c khi suáº¥t cÅ© káº¿t thÃºc
       AND s.end_time   > p_start;   -- vÃ  káº¿t thÃºc sau khi suáº¥t cÅ© báº¯t Ä‘áº§u
    RETURN v_cnt;
  END fn_check_room_overlap;

  PROCEDURE sp_add_showtime(
    p_movie_id NUMBER, p_room_id NUMBER, p_start TIMESTAMP, p_end TIMESTAMP,
    p_base_price NUMBER, p_showtime_id OUT NUMBER) IS
  BEGIN
    IF fn_check_room_overlap(p_room_id, p_start, p_end) > 0 THEN
      RAISE_APPLICATION_ERROR(-20050, 'Phong da co suat chieu trung gio');
    END IF;

    INSERT INTO showtime(movie_id, room_id, start_time, end_time, base_price, status)
    VALUES (p_movie_id, p_room_id, p_start, p_end, p_base_price, 'OPEN')
    RETURNING showtime_id INTO p_showtime_id;
    COMMIT;
    -- LÆ°u Ã½: trigger trg_showtime_seed (file 04) sáº½ tá»± Ä‘á»™ng cháº¡y ngay sau INSERT
    -- nÃ y Ä‘á»ƒ sinh sáºµn cÃ¡c dÃ²ng gháº¿ trá»‘ng cho suáº¥t chiáº¿u vá»«a táº¡o.
  END sp_add_showtime;

END pkg_booking;

-- Náº¿u Oracle bÃ¡o thiáº¿u quyá»n táº¡o MLE MODULE, xem láº¡i file 00

-- 1. Táº¡o MLE MODULE chá»©a mÃ£ JavaScript
CREATE OR REPLACE MLE MODULE mod_password_check
LANGUAGE JAVASCRIPT AS

export function isStrongPassword(pwd) {
  if (!pwd || pwd.length < 8) return 0;
  const hasUpper = /[A-Z]/.test(pwd);
  const hasLower = /[a-z]/.test(pwd);
  const hasDigit = /[0-9]/.test(pwd);
  return (hasUpper && hasLower && hasDigit) ? 1 : 0;
}
/

-- 2. Táº¡o Function PL/SQL gá»i hÃ m JavaScript trÃªn qua MLE

RETURN NUMBER
AS MLE MODULE mod_password_check
SIGNATURE 'isStrongPassword(string)';

CREATE OR REPLACE PROCEDURE sp_register_customer(
  p_full_name IN VARCHAR2, p_email IN VARCHAR2,
  p_password  IN VARCHAR2, p_customer_id OUT NUMBER) IS
BEGIN
  IF fn_is_strong_password(p_password) = 0 THEN
    RAISE_APPLICATION_ERROR(-20060,
      'Mat khau phai co it nhat 8 ky tu, gom chu hoa, chu thuong va so');
  END IF;
  INSERT INTO customer(full_name, email, password_hash)
  VALUES (p_full_name, p_email, STANDARD_HASH(p_password, 'SHA256'))
  RETURNING customer_id INTO p_customer_id;
  COMMIT;
EXCEPTION
  WHEN DUP_VAL_ON_INDEX THEN
    RAISE_APPLICATION_ERROR(-20061, 'Email da duoc dang ky');
END sp_register_customer;

-- SÆ¡ Ä‘á»“ gháº¿ cho giao diá»‡n, view + hÃ m thá»‘ng kÃª doanh thu/láº¥p Ä‘áº§y
-- (má»¥c 3.11.a vÃ  3.11.e bÃ¡o cÃ¡o).

-- 3.11.a. SÆ¡ Ä‘á»“ gháº¿ â€” giao diá»‡n chá»‰ Ä‘á»c view nÃ y, khÃ´ng Ä‘á»c tháº³ng SHOWTIME_SEAT
CREATE OR REPLACE VIEW v_seat_map AS
SELECT ss.showtime_id, s.seat_id, s.row_label, s.seat_no, s.seat_type,
       CASE WHEN ss.status = 'HELD' AND ss.hold_expires_at < SYSTIMESTAMP
            THEN 'AVAILABLE' ELSE ss.status END AS status
  FROM showtime_seat ss JOIN seat s ON s.seat_id = ss.seat_id;

-- 3.11.e. Thá»‘ng kÃª doanh thu vÃ  tá»‰ lá»‡ láº¥p Ä‘áº§y (cho quáº£n trá»‹)
CREATE OR REPLACE VIEW v_revenue_by_movie AS
SELECT m.movie_id, m.title, TRUNC(st.start_time) AS show_date,
       COUNT(*) AS tickets_sold, SUM(t.price) AS gross_revenue
  FROM ticket t
  JOIN booking  b  ON b.booking_id  = t.booking_id AND b.status = 'PAID'
  JOIN showtime st ON st.showtime_id = t.showtime_id
  JOIN movie    m  ON m.movie_id     = st.movie_id
 GROUP BY m.movie_id, m.title, TRUNC(st.start_time);

CREATE OR REPLACE FUNCTION fn_occupancy_rate(p_showtime_id NUMBER) RETURN NUMBER IS
  v_rate NUMBER;
BEGIN
  SELECT ROUND(100 * COUNT(CASE WHEN status = 'SOLD' THEN 1 END) / NULLIF(COUNT(*), 0), 1)
    INTO v_rate FROM showtime_seat WHERE showtime_id = p_showtime_id;
  RETURN NVL(v_rate, 0);
END;
